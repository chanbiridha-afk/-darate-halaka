// ============================================================
// قواعد أمان Firestore — منصة إدارة حلقة التحفيظ
// انسخوا هذا المحتوى بالكامل إلى: Firebase Console → Firestore Database → Rules → ثم اضغطوا Publish
// ============================================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function myProfile() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function myRole() { return signedIn() ? myProfile().role : null; }
    function isAdmin() { return myRole() == "admin"; }
    function isTeacher() { return myRole() == "teacher"; }
    function isParent() { return myRole() == "parent"; }
    function myGroups() { return isTeacher() ? myProfile().assignedGroupIds : []; }
    function studentOf(sid) { return get(/databases/$(database)/documents/students/$(sid)).data; }

    // ملفات المستخدمين (الأدوار)
    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow create: if signedIn() && request.auth.uid == uid;
      allow update: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow delete: if isAdmin();
    }

    // السنوات الدراسية والأفواج: يقرؤها كل الطاقم، يديرها المدير فقط
    match /years/{id} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }
    match /groups/{id} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    // المتعلّمون
    // ملاحظة: يُسمح لأي ولي أمر مسجَّل بقراءة وثائق المتعلمين (وليس فقط أبناءه) لتمكين
    // البحث بالرقم التعريفي عند عملية الربط الأولى؛ الرقم التعريفي + تاريخ الميلاد يُشكّلان
    // تحققًا ثنائيًا كافيًا لسياق مدرسة صغيرة موثوقة. الكتابة تبقى محصورة بالكامل.
    match /students/{id} {
      allow read: if isAdmin()
                  || (isTeacher() && resource.data.groupId in myGroups())
                  || isParent();
      allow create: if isAdmin();
      allow update: if isAdmin()
                    || (isTeacher() && resource.data.groupId in myGroups())
                    || (isParent() && request.auth.uid in request.resource.data.parentUids);
      allow delete: if isAdmin();
    }

    // الجلسات
    match /sessions/{id} {
      allow read: if isAdmin()
                  || (isTeacher() && studentOf(resource.data.studentId).groupId in myGroups())
                  || (isParent() && request.auth.uid in studentOf(resource.data.studentId).parentUids);
      allow create: if isAdmin() || isTeacher();
      allow update, delete: if isAdmin() || isTeacher();
    }

    // الاختبارات والفروض والتقويم المستمر
    match /exams/{id} {
      allow read: if isAdmin()
                  || (isTeacher() && studentOf(resource.data.studentId).groupId in myGroups())
                  || (isParent() && request.auth.uid in studentOf(resource.data.studentId).parentUids);
      allow write: if isAdmin() || isTeacher();
    }

    // الاشتراكات المالية: للمدير فقط، مع سماح القراءة لولي الأمر المرتبط بابنه
    match /payments/{id} {
      allow read: if isAdmin()
                  || (isParent() && request.auth.uid in studentOf(resource.data.studentId).parentUids);
      allow write: if isAdmin();
    }
  }
}
